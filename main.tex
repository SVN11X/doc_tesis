% 14. MACROS PARA FIGURAS CON FORMATO UTEM
% =========================================================================================

% -----------------------------------------------------------------------------------------
% Figura con fuente bibliográfica.
% Uso:
% \utemFigure{ruta/imagen.png}{0.8\textwidth}{Título}{claveBib}{fig:etiqueta}
% -----------------------------------------------------------------------------------------
\newcommand{\utemFigure}[5]{
  \begin{figure}[H]
    \centering

    \setlength{\fboxsep}{6pt}
    \fbox{%
      \includegraphics[width=#2]{#1}%
    }

    \vspace{4pt}

    \caption{#3}
    \label{#5}

    \vspace{2pt}

    {\footnotesize\textit{Fuente: \citep{#4}.}}
  \end{figure}
}

% -----------------------------------------------------------------------------------------
% Figura de elaboración propia.
% Uso:
% \utemFigurePropia{ruta/imagen.png}{0.8\textwidth}{Título}{fig:etiqueta}
% -----------------------------------------------------------------------------------------
\newcommand{\utemFigurePropia}[4]{
  \begin{figure}[H]
    \centering

    \setlength{\fboxsep}{6pt}
    \fbox{%
      \includegraphics[width=#2]{#1}%
    }

    \vspace{4pt}

    \caption{#3}
    \label{#4}

    \vspace{2pt}

    {\footnotesize\textit{Fuente: Elaboración propia.}}
  \end{figure}
}

% -----------------------------------------------------------------------------------------
% Varias imágenes con fuentes bibliográficas.
% Uso:
% \begin{utemMultiFigure}{Título general}{fig:etiqueta}
%   \addutem{0.45}{ruta/img1.png}{Título 1}{claveBib1}
%   \hfill
%   \addutem{0.45}{ruta/img2.png}{Título 2}{claveBib2}
% \end{utemMultiFigure}
% -----------------------------------------------------------------------------------------
\newenvironment{utemMultiFigure}[2]{
  \def\utemMFcapt{#1}%
  \def\utemMFlab{#2}%
  \begin{figure}[H]
    \centering
    \setlength{\fboxsep}{6pt}
    \providecommand{\addutem}{}
    \renewcommand{\addutem}[4]{
      \begin{subfigure}[b]{##1\textwidth}
        \centering
        \fbox{%
          \includegraphics[width=\textwidth]{##2}%
        }
        \caption{##3}
        \vspace{2pt}

        {\footnotesize\textit{Fuente: \citep{##4}.}}
      \end{subfigure}
    }
}{
    \vspace{4pt}
    \caption{\utemMFcapt}
    \label{\utemMFlab}
  \end{figure}
}

% -----------------------------------------------------------------------------------------
% Varias imágenes de elaboración propia.
% Uso:
% \begin{utemMultiFigurePropia}{Título general}{fig:etiqueta}
%   \addutem{0.45}{ruta/img1.png}{Título 1}
%   \hfill
%   \addutem{0.45}{ruta/img2.png}{Título 2}
% \end{utemMultiFigurePropia}
% -----------------------------------------------------------------------------------------
\newenvironment{utemMultiFigurePropia}[2]{
  \def\utemMFcapt{#1}%
  \def\utemMFlab{#2}%
  \begin{figure}[H]
    \centering
    \setlength{\fboxsep}{6pt}
    \providecommand{\addutem}{}
    \renewcommand{\addutem}[3]{
      \begin{subfigure}[b]{##1\textwidth}
        \centering
        \fbox{%
          \includegraphics[width=\textwidth]{##2}%
        }
        \caption{##3}
      \end{subfigure}
    }
}{
    \vspace{4pt}
    \caption{\utemMFcapt}
    \label{\utemMFlab}

    \vspace{2pt}

    {\footnotesize\textit{Fuente: Elaboración propia.}}
  \end{figure}
}
