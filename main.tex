% ===== Figura tipo UTEM =====
\newcommand{\utemFigure}[5]{%
  \begin{figure}[H]
    \centering
    \setlength{\fboxsep}{6pt}
    \fbox{\includegraphics[width=#2]{#1}}
    \caption{#3}
    \vspace{6pt}
    \footnotesize \textit{\textbf{Fuente: \cite{#4}. Enlace: \citefield{#4}{url}}}
    \label{#5}
  \end{figure}
}

% ===== Figura tipo UTEM para elaboracion propia=====
\newcommand{\utemFigurePropia}[4]{%
  \begin{figure}[H]
    \centering
    \setlength{\fboxsep}{6pt}
    \fbox{\includegraphics[width=#2]{#1}}
    \caption{#3}
    \vspace{6pt}
    \footnotesize \textit{Elaboración Propia}
    \label{#4}
  \end{figure}
}

% ===== Entorno UTEM Dinámico para Múltiples Imágenes =====
\newenvironment{utemMultiFigure}[2]{%
    % 1. Guardamos los argumentos en comandos temporales
    \def\tempCaption{#1}% 
    \def\tempLabel{#2}%
    
    \begin{figure}[H]
    \centering
    \setlength{\fboxsep}{6pt}
    
    % Definimos el comando interno (usamos renewcommand para evitar errores de duplicado)
    \newcommand{\addutem}[5]{%
        \begin{subfigure}[b]{##1\textwidth}
            \centering
            \fbox{\includegraphics[width=\textwidth]{##2}}
            \caption{##3}
            \vspace{6pt}
            \footnotesize \textit{\textbf{Fuente: \cite{##4}. Enlace: \citefield{##4}{url}}}
            \label{##5}
        \end{subfigure}
    }
}{%
    % 2. Aquí usamos las memorias temporales en lugar de #1 y #2
    \vspace{8pt}
    \caption{\tempCaption}
    \label{\tempLabel}
    \end{figure}
}

% ===== Entorno UTEM Dinámico para Múltiples Imágenes Propia =====
