\vspace*{2cm}

\begin{flushright}
\textit{A mi familia,} \\
\textit{por su apoyo incondicional y por enseñarme que la perseverancia} \\
\textit{es la llave que abre las puertas de los sueños más grandes.} \\[1.5cm]

\textit{A todos aquellos que, con su dedicación y conocimiento,} \\
\textit{contribuyen a que la robótica sea una herramienta} \\
\textit{para transformar la realidad.}
\end{flushright}
